    // Smooth navigation section scrolling

    }

    // Handle Hero Quick Check Form Submission

    function handleQuickCheck(e) {

      e.preventDefault();

      const carModel = document.getElementById('carModel').value;

      const primaryIssue = document.getElementById('primaryIssue').value;

      if (carModel) {

        document.getElementById('vehicleMakeModel').value = carModel;

      }

      scrollToSection('inquiry');

      nextWizardStep(2);

    }

    // Step Wizard navigation logic

    function nextWizardStep(stepNumber) {

      // Validate inputs before moving forward

      if (stepNumber === 2) {

        const name = document.getElementById('ownerName').value.trim();

        const phone = document.getElementById('ownerPhone').value.trim();

        const vehicle = document.getElementById('vehicleMakeModel').value.trim();

        if (!name || !phone || !vehicle) {

          alert('Please fill in your Name, Phone Number, and Vehicle Model before proceeding.');

          return;

        }

      }

      if (stepNumber === 3) {

        const desc = document.getElementById('problemDescription').value.trim();

        if (!desc) {

          alert('Please provide a brief description of the problem.');

          return;

        }

      }

      if (stepNumber === 4) {

        updateSummary();

      }

      // Toggle form step visibility

      const steps = document.querySelectorAll('.wizard-step-content');

      steps.forEach(step => step.style.display = 'none');

      const targetStep = document.getElementById('wizardStep' + stepNumber);

      if (targetStep) {

        targetStep.style.display = 'block';

      }

      // Update Top Step Indicator styling

      const stepIndicators = document.querySelectorAll('.wizard-steps .step');

      stepIndicators.forEach((indicator, idx) => {

        if (idx + 1 === stepNumber) {

          indicator.classList.add('active');

        } else {

          indicator.classList.remove('active');

        }

      });

    }

    // Populate data for step 4 summary preview

    function updateSummary() {

      const name = document.getElementById('ownerName').value || '--';

      const phone = document.getElementById('ownerPhone').value || '--';

      const vehicle = document.getElementById('vehicleMakeModel').value || '--';

      const selectedSymptoms = Array.from(document.querySelectorAll('input[name="symptoms"]:checked'))

                                   .map(cb => cb.value);

      const desc = document.getElementById('problemDescription').value;

      let symptomsText = selectedSymptoms.join(', ');

      if (desc) {

        symptomsText += (symptomsText ? ' | Note: ' : '') + desc;

      }

      document.getElementById('summaryName').textContent = name;

      document.getElementById('summaryPhone').textContent = phone;

      document.getElementById('summaryVehicle').textContent = vehicle;

      document.getElementById('summarySymptoms').textContent = symptomsText || 'None specified';

    }
